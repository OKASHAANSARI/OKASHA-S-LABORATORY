CREATE TABLE PERSON( 
   driver_id VARCHAR2(20) PRIMARY KEY, 
   name VARCHAR2(40), 
   address VARCHAR2(100)
);

CREATE TABLE CAR ( 
   regno VARCHAR2(20) PRIMARY KEY, 
   model VARCHAR2(40), 
   year NUMBER(4)
);

CREATE TABLE ACCIDENT( 
   report_number NUMBER PRIMARY KEY, 
   accd_date DATE, 
   location VARCHAR2(100)
);

CREATE TABLE OWNS( 
   driver_id VARCHAR2(20), 
   regno VARCHAR2(20), 

   PRIMARY KEY(driver_id, regno), 

   FOREIGN KEY (driver_id) REFERENCES PERSON (driver_id),
   FOREIGN KEY (regno) REFERENCES CAR (regno)
);

CREATE TABLE PARTICIPATED( 
   driver_id VARCHAR2(20), 
   regno VARCHAR2(20), 
   report_number NUMBER, 
   damage_amount NUMBER, 

   PRIMARY KEY (driver_id, regno, report_number),

   FOREIGN KEY (driver_id) REFERENCES PERSON (driver_id),
   FOREIGN KEY (regno) REFERENCES CAR (regno),
   FOREIGN KEY (report_number) REFERENCES ACCIDENT (report_number)
);

SELECT COUNT (DISTINCT o.driver_id) as Total_people
FROM OWNS o
JOIN  PARTICIPATED p
  on o.driver_id = p.driver_id 
  AND o.regno = p.regno
JOIN ACCIDENT a
  on a.report_number = p.report_number
WHERE a.accd_date >= DATE '2008-01-01'
AND a.accd_date < DATE '2009-01-01';


SELECT COUNT(DISTINCT p.report_number) AS number_of_accidents
FROM PARTICIPATED p
JOIN CAR c
    ON p.regno = c.regno
WHERE c.model = 'Honda City';
